```python
#!/usr/bin/env python
"""Aggregate OPRS cache → data/processed/prc.parquet (D-32)."""
from __future__ import annotations

import sys
from pathlib import Path

from fairhaven_tax.ingest.oprs.parse_m4 import parse_m4
from fairhaven_tax.ingest.oprs.parse_prc_pdf import parse_prc_pdf
from fairhaven_tax.ingest.oprs.parse_ch75_pdf import parse_ch75_pdf
from fairhaven_tax.ingest.oprs.parse_taxlist_pdf import parse_taxlist_pdf
from fairhaven_tax.persist.parquet_io import write_parquet


def main() -> int:
    cache_root = Path("data/raw/oprs_prc")
    if not cache_root.exists():
        print(f"ERROR: missing {cache_root}; run datasets/collect_oprs.py first",
              file=sys.stderr)
        return 2

    rows = []
    for parcel_dir in sorted(cache_root.iterdir()):
        if not parcel_dir.is_dir(): continue
        row = {"pams_pin": parcel_dir.name}
        m4 = parcel_dir / "m4.html"
        if m4.exists():
            row.update(parse_m4(m4))
        prc = parcel_dir / "prc.pdf"
        if prc.exists():
            row.update(parse_prc_pdf(prc))
        # ch75, taxlist...
        rows.append(row)

    import pandas as pd
    df = pd.DataFrame(rows)
    write_parquet(df, "data/processed/prc.parquet")
    print(f"Wrote {len(df)} parcel rows → data/processed/prc.parquet")
    if len(df) == 0:
        print("ERROR: zero parcels parsed — refusing to declare success",
              file=sys.stderr)
        return 2
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
