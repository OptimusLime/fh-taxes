```python
# §3.1 COD
def cod(ratio):
    ratio = np.asarray(ratio, dtype=float)
    ratio = ratio[~np.isnan(ratio)]
    med = np.median(ratio)
    return (np.mean(np.abs(ratio - med)) / med) * 100.0

def cod_met(x): return 5.0 <= x <= 15.0

# §3.2 PRD
def prd(assessed, sale_price):
    a = np.asarray(assessed, dtype=float); p = np.asarray(sale_price, dtype=float)
    mask = ~(np.isnan(a) | np.isnan(p))
    a, p = a[mask], p[mask]
    ratio = a / p
    return ratio.mean() / np.average(ratio, weights=p)

def prd_met(x): return 0.98 <= x <= 1.03

# §3.3 PRB (statsmodels OLS port of R lm)
def prb(assessed, sale_price):
    a = np.asarray(assessed, dtype=float); p = np.asarray(sale_price, dtype=float)
    mask = ~(np.isnan(a) | np.isnan(p))
    a, p = a[mask], p[mask]
    ratio = a / p
    med = np.median(ratio)
    lhs = (ratio - med) / med
    rhs = np.log(((a / med) + p) * 0.5) / np.log(2)
    rhs = sm.add_constant(rhs)
    model = sm.OLS(lhs, rhs).fit()
    return float(model.params[1])

def prb_met(x): return -0.05 <= x <= 0.05
```

       ```python
       def trim_outliers_iqr(ratio: np.ndarray, k: float = 3.0) -> np.ndarray:
           r = np.asarray(ratio, dtype=float)
           r = r[~np.isnan(r)]
           if len(r) < 4:
               return r  # cannot compute IQR meaningfully
           q1, q3 = np.percentile(r, [25, 75])
           iqr = q3 - q1
           med = np.median(r)
           lo, hi = med - k * iqr, med + k * iqr
           return r[(r >= lo) & (r <= hi)]
       ```

       ```python
           year_min = year_min or constants.CDF_TEST_YEAR_MIN

               (modiv_history['sale_price'].notna()) &

           n = len(df)
           if n < 2:
               return {'cohort_tag': 'full_sample', 'n': n, 'median_ratio': None,
                       'cod': None, 'prd': None, 'prb': None,
                       'cod_pass': None, 'prd_pass': None, 'prb_pass': None,
                       'n_warning': True, 'year_min': year_min, 'year_max': year_max}
           cod_val = cod(df['ratio'].values, trim=True)
           prd_val = prd(df['sale_assessment'].astype(float).values,
                         df['sale_price'].astype(float).values)
           prb_val = prb(df['sale_assessment'].astype(float).values,
                         df['sale_price'].astype(float).values)
           return {

               'median_ratio': Decimal(str(round(float(np.median(df['ratio'])), 6))),
               'cod': Decimal(str(round(cod_val, 4))),
       ```
