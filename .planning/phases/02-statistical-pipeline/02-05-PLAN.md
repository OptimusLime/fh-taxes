       def hedonic_per_parcel_delta(parcels, total_levy=None, tax_rate_per_100=None):
           if total_levy is None:
               total_levy = float(constants.TOTAL_LEVY)
           if tax_rate_per_100 is None:
               tax_rate_per_100 = float(constants.TAX_RATE_PER_HUNDRED)
           p = parcels.copy()
           p['_etv'] = p['estimated_true_value'].astype(float)
           p['_av']  = p['assessed_value'].astype(float)
           # Apply the calibration so fair_bill aggregates to actual levy.
           share = p['_etv'] / p['_etv'].sum()
           p['fair_bill']   = share * total_levy
           p['actual_bill'] = p['_av'] * (tax_rate_per_100 / 100.0)
           p['delta_dollars'] = p['actual_bill'] - p['fair_bill']
           # CALC-01 invariant
           agg_delta = p['delta_dollars'].sum()
           if abs(agg_delta) > 0.01 * total_levy:
               raise AssertionError(
                   f"Σ delta_dollars = {agg_delta:.2f}; expected ≈ 0 within 1% of total_levy "
                   f"({0.01 * total_levy:.2f}). Check assessed_value source.")
           return p[['pams_pin', 'fair_bill', 'actual_bill', 'delta_dollars']].assign(
               # cast back to Decimal at output boundary (S5)
               fair_bill=lambda d: d['fair_bill'].map(lambda x: Decimal(str(round(x, 2)))),
