       def _aggregate_parcel(parcel_dir: Path) -> tuple[dict | None, list[dict]]:
           """Return (parcel_row, sale_rows) or (None, []) if m4 missing."""
           m4_path = parcel_dir / "m4.html"
           if not m4_path.exists():
               return None, []
           row: dict = {"pams_pin": parcel_dir.name}
           row.update(parse_m4(m4_path))

           prc_path = parcel_dir / "prc.pdf"
           if prc_path.exists():
               row.update(parse_prc_pdf(prc_path))
           else:
               row.update({k: None for k in PRC_PDF_FIELDS})

           ch75_path = parcel_dir / "ch75.pdf"
           if ch75_path.exists():
               row.update(parse_ch75_pdf(ch75_path))
           else:
               row.update({k: None for k in CH75_FIELDS})

           taxlist_paths = sorted(parcel_dir.glob("taxlist_*.pdf"))
           if taxlist_paths:
               row.update(parse_taxlist_pdf(taxlist_paths[-1]))  # latest year
           else:
               row.update({k: None for k in TAXLIST_FIELDS})

           sale_rows: list[dict] = []
           for sr_path in sorted(parcel_dir.glob("sr_*.html")):
               if sr_path.suffix.endswith(".no_sale"):
                   continue
               sr = parse_sr(sr_path)
               if sr is not None:
                   sale_rows.append(sr)
           return row, sale_rows

       def main() -> int:
           if not CACHE_ROOT.exists():
               print(f"ERROR: missing {CACHE_ROOT}; run datasets/collect_oprs.py first", file=sys.stderr)
               return 2

           parcel_rows: list[dict] = []
           sale_rows_all: list[dict] = []
           parcels_seen = 0
           parcels_skipped = 0

           for d in sorted(CACHE_ROOT.iterdir()):
               if not d.is_dir():
                   continue
               parcels_seen += 1
               row, sale_rows = _aggregate_parcel(d)
               if row is None:
                   parcels_skipped += 1
                   continue
               parcel_rows.append(row)
               sale_rows_all.extend(sale_rows)

           if not parcel_rows:
               print("ERROR: zero parcels parsed — refusing to declare success", file=sys.stderr)
               return 2

           df_prc = pd.DataFrame(parcel_rows)
           write_parquet(df_prc, str(OUT_PRC))
           print(f"Wrote {len(df_prc)} rows → {OUT_PRC} (skipped {parcels_skipped} parcels missing m4)")

           # Cross-validation
           diffs = _cross_validate_sqft(df_prc)
           if len(diffs) > 0:
               write_parquet(diffs, str(OUT_DIFFS))
               print(f"Wrote {len(diffs)} sqft-diff rows → {OUT_DIFFS}")

           # Sales enrichment (D-33)
           if SALES_PATH.exists() and sale_rows_all:
               df_sr = pd.DataFrame(sale_rows_all)
               sales = pd.read_parquet(SALES_PATH)
               # Merge on (parcel_pin, sale_date). If sr provides extra columns, append them.
               sales_enriched = sales.merge(
                   df_sr,
                   on=["parcel_pin", "sale_date"],
                   how="left",
                   suffixes=("", "_sr"),
               )
               write_parquet(sales_enriched, str(SALES_PATH))
               print(f"Enriched {len(sales)} sales with {len(df_sr)} sr rows → {SALES_PATH}")
           elif not SALES_PATH.exists():
               print(f"WARN: {SALES_PATH} not found — skipping sales enrichment", file=sys.stderr)

           return 0


       if __name__ == "__main__":
           sys.exit(main())
